## Production Notes

### Scene Generation Automation

```python
#!/usr/bin/env python3
# generate-movie.py - Automated movie generation

import os
from moviepy.editor import *
from elevenlabs import generate_voice
from stable_diffusion import generate_image
import manim

def create_scene(scene_id, narration_text, visual_prompts):
    """Generate a single scene"""
    
    # Generate voice narration
    audio = generate_voice(narration_text, voice="professional_male")
    
    # Generate visuals
    images = [generate_image(prompt) for prompt in visual_prompts]
    
    # Combine into video clip
    clips = []
    for img in images:
        clip = ImageClip(img, duration=len(audio) / len(images))
        clips.append(clip)
    
    video = concatenate_videoclips(clips)
    video = video.set_audio(audio)
    
    return video

# Generate all scenes
scenes = []
scenes.append(create_scene("1.1", "When vulnerability strikes...", ["medical", "code", "math"]))
# ... etc

# Combine all scenes
final_movie = concatenate_videoclips(scenes)
final_movie.write_videofile(
    "sovereignty-architecture-movie.mp4",
    fps=60,
    codec='libx264',
    audio_codec='aac',
    bitrate='8000k'
)
```
