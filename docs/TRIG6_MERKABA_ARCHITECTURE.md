import math

class TRIG6:
    """TRIG6 family with Merkaba geometry awareness"""
    
    @staticmethod
    def sin(theta):
        """Ascending function - upward tetrahedron vertex"""
        return math.sin(theta)
    
    @staticmethod
    def cos(theta):
        """Grounding function - upward tetrahedron base"""
        return math.cos(theta)
    
    @staticmethod
    def tan(theta):
        """Explosive ratio - upward tetrahedron edge"""
        return math.tan(theta) if math.cos(theta) != 0 else float('inf')
    
    @staticmethod
    def csc(theta):
        """Inverse rise - downward tetrahedron (mirror SIN)"""
        return 1 / math.sin(theta) if math.sin(theta) != 0 else float('inf')
    
    @staticmethod
    def sec(theta):
        """Inverse ground - downward tetrahedron (mirror COS)"""
        return 1 / math.cos(theta) if math.cos(theta) != 0 else float('inf')
    
    @staticmethod
    def cot(theta):
        """Inverse slope - downward tetrahedron (mirror TAN)"""
        return 1 / math.tan(theta) if math.tan(theta) != 0 else float('inf')
    
    @classmethod
    def merkaba_balance(cls, theta):
        """Calculate the balance point between upward and downward forces"""
        upward = cls.sin(theta) + cls.cos(theta) + cls.tan(theta)
        downward = cls.csc(theta) + cls.sec(theta) + cls.cot(theta)
        return (upward - downward) / 2  # The intersection
